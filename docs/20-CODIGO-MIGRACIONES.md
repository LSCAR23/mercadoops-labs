# 20 — Código íntegro de las migraciones (para consulta en Markdown)

Este documento replica literalmente las migraciones incluidas en el starter. **La fuente ejecutable es `db/alembic/versions/*.py`**; este `.md` es una referencia cómoda para Cursor. Ningún bloque a continuación resuelve los ejercicios SQL: solo define estructura de almacenamiento e índices operacionales mínimos.

**IMPORTANTE:** si el esquema evoluciona después de aplicar estas revisiones, añade una nueva migración. No edites una revisión histórica en un entorno ya migrado. Ejecuta `uv run alembic upgrade head` desde la raíz.

"""Core tenants, customers and categories.

Revision ID: 0001_core
Revises:
"""

revision = "0001_core"
down_revision = None
branch_labels = None
depends_on = None

    op.execute("""
        CREATE TABLE core.tenants (
            id UUID PRIMARY KEY,
            slug TEXT NOT NULL UNIQUE,
            display_name TEXT NOT NULL,
            created_at TIMESTAMPTZ NOT NULL DEFAULT now(),

        )
    """)
    op.execute("""
        CREATE TABLE core.customers (
            id UUID PRIMARY KEY,
            tenant_id UUID NOT NULL REFERENCES core.tenants(id),
            email TEXT NOT NULL,
            full_name TEXT NOT NULL,
            country_code CHAR(2) NOT NULL,
            acquisition_channel TEXT NOT NULL,
            acquired_at TIMESTAMPTZ NOT NULL,
            created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
            CONSTRAINT uq_customers_tenant_id UNIQUE (tenant_id, id),
            CONSTRAINT uq_customers_tenant_email UNIQUE (tenant_id, email),
            CONSTRAINT ck_customer_channel CHECK (
                acquisition_channel IN ('organic', 'paid', 'social', 'referral', 'direct'))
        )
    """)
    op.execute("""
        CREATE TABLE core.categories (
            id UUID PRIMARY KEY,
            tenant_id UUID NOT NULL REFERENCES core.tenants(id),
            name TEXT NOT NULL,
            CONSTRAINT uq_categories_tenant_id UNIQUE (tenant_id, id),
            CONSTRAINT uq_categories_tenant_name UNIQUE (tenant_id, name)
        )
    """)


def downgrade() -> None:
    op.execute("DROP TABLE core.categories")
    op.execute("DROP TABLE core.customers")
    op.execute("DROP TABLE core.tenants")
    op.execute("DROP SCHEMA analytics")
    op.execute("DROP SCHEMA ops")
    op.execute("DROP SCHEMA events")
    op.execute("DROP SCHEMA core")
```

"""Product catalog and inventory.

Revision ID: 0002_catalog
Revises: 0001_core
"""
from alembic import op

revision = "0002_catalog"
down_revision = "0001_core"
branch_labels = None
depends_on = None


def upgrade() -> None:
    op.execute("""
        CREATE TABLE core.products (
            id UUID PRIMARY KEY,
            tenant_id UUID NOT NULL REFERENCES core.tenants(id),
            category_id UUID NOT NULL,
            sku TEXT NOT NULL,
            name TEXT NOT NULL,
            unit_price NUMERIC(12, 2) NOT NULL,
            unit_cost NUMERIC(12, 2) NOT NULL,
            is_active BOOLEAN NOT NULL DEFAULT TRUE,
            created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
            CONSTRAINT uq_products_tenant_id UNIQUE (tenant_id, id),
            CONSTRAINT uq_products_tenant_sku UNIQUE (tenant_id, sku),
            CONSTRAINT fk_product_category FOREIGN KEY (tenant_id, category_id)
                REFERENCES core.categories(tenant_id, id),
            CONSTRAINT ck_product_prices CHECK (
                unit_price > 0 AND unit_cost >= 0 AND unit_cost <= unit_price)
        )
    """)
    op.execute("""
        CREATE TABLE core.inventory (
            tenant_id UUID NOT NULL,
            product_id UUID NOT NULL,
            available_qty INTEGER NOT NULL DEFAULT 0,
            reserved_qty INTEGER NOT NULL DEFAULT 0,
            version INTEGER NOT NULL DEFAULT 0,
            updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),
            PRIMARY KEY (tenant_id, product_id),
            CONSTRAINT fk_inventory_product FOREIGN KEY (tenant_id, product_id)
                REFERENCES core.products(tenant_id, id),
            CONSTRAINT ck_inventory_nonnegative CHECK (
                available_qty >= 0 AND reserved_qty >= 0 AND version >= 0)
        )
    """)
    op.execute("""
        CREATE TABLE core.stock_movements (
            id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
            tenant_id UUID NOT NULL,
            product_id UUID NOT NULL,
            qty_delta INTEGER NOT NULL,
            reason TEXT NOT NULL,
            recorded_at TIMESTAMPTZ NOT NULL DEFAULT now(),
            CONSTRAINT fk_stock_product FOREIGN KEY (tenant_id, product_id)
                REFERENCES core.products(tenant_id, id),
            CONSTRAINT ck_stock_delta CHECK (qty_delta <> 0),
            CONSTRAINT ck_stock_reason CHECK (
                reason IN ('opening', 'sale', 'return', 'adjustment', 'restock'))
        )
    """)


def downgrade() -> None:
    op.execute("DROP TABLE core.stock_movements")
    op.execute("DROP TABLE core.inventory")
    op.execute("DROP TABLE core.products")
```

            CONSTRAINT ck_payment_status CHECK (
                payment_status IN ('pending', 'authorized', 'captured', 'failed', 'voided')),
